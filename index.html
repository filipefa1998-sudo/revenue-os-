Revenue Compliance Index™ (RCI)

The moat. 0-100 score measuring sales discipline.

The 5 Weighted Categories

Category Weight What It Measures
Stage Discipline 30% % compliant transitions, gate violations, non-compliant days
Follow-Up Adherence 25% % on-time follow-ups, SLA adherence
Activity Velocity 20% Actual vs expected touch frequency per stage
Data Hygiene 15% Required fields, notes quality, duplicates
Forecast Accuracy 10% Forecasted vs actual variance (quarterly)

The Formula

RCI = (stage_score × 0.3) +
(followup_score × 0.25) +
(velocity_score × 0.2) +
(hygiene_score × 0.15) +
(forecast_score × 0.1)

Database Schema (MVP)

9 tables only. Minimal viable structure.

Core Tables

-- ORGANIZATIONS
CREATE TABLE organizations (
  id UUID PRIMARY KEY,
  name TEXT NOT NULL,
  motto TEXT,
  created_at TIMESTAMP DEFAULT NOW()
);

-- USERS
CREATE TABLE users (
  id UUID PRIMARY KEY,
  org_id UUID REFERENCES organizations(id),
  email TEXT NOT NULL,
  role TEXT CHECK (role IN ('admin','manager','rep','coo')),
  created_at TIMESTAMP DEFAULT NOW()
);

-- PIPELINES
CREATE TABLE pipelines (
  id UUID PRIMARY KEY,
  org_id UUID REFERENCES organizations(id),
  external_pipeline_id TEXT NOT NULL,
  name TEXT NOT NULL
);

-- STAGES
CREATE TABLE stages (
  id UUID PRIMARY KEY,
  pipeline_id UUID REFERENCES pipelines(id),
  external_stage_id TEXT NOT NULL,
  name TEXT NOT NULL,
  order_index INT NOT NULL
);

-- STAGE REQUIREMENTS (The Moat)
CREATE TABLE stage_requirements (
  id UUID PRIMARY KEY,
  stage_id UUID REFERENCES stages(id),
  enforcement_mode TEXT CHECK (enforcement_mode IN ('HARD','SOFT')),
  sla_followup_hours INT DEFAULT 48,
  required_fields JSONB DEFAULT '[]',
  required_activities JSONB DEFAULT '[]'
);

-- DEALS
CREATE TABLE deals (
  id UUID PRIMARY KEY,
  org_id UUID REFERENCES organizations(id),
  external_deal_id TEXT NOT NULL,
  owner_user_id UUID REFERENCES users(id),
  stage_id UUID REFERENCES stages(id),
  amount NUMERIC(15,2),
  close_date DATE,
  health_status TEXT CHECK (health_status IN ('GREEN','YELLOW','RED'))
);

-- ACTIVITIES
CREATE TABLE activities (
  id UUID PRIMARY KEY,
  org_id UUID REFERENCES organizations(id),
  deal_id UUID REFERENCES deals(id),
  user_id UUID REFERENCES users(id),
  type TEXT CHECK (type IN ('email','meeting','call','note')),
  occurred_at TIMESTAMP NOT NULL
);

-- COMPLIANCE SCORES
CREATE TABLE compliance_scores (
  id UUID PRIMARY KEY,
  org_id UUID REFERENCES organizations(id),
  user_id UUID REFERENCES users(id),
  period_start DATE NOT NULL,
  period_end DATE NOT NULL,
  rci_total NUMERIC(6,2) NOT NULL,
  computed_at TIMESTAMP DEFAULT NOW()
);

-- AI SUGGESTIONS
CREATE TABLE ai_suggestions (
  id UUID PRIMARY KEY,
  org_id UUID REFERENCES organizations(id),
  deal_id UUID REFERENCES deals(id),
  text TEXT NOT NULL,
  sources JSONB DEFAULT '[]',
  created_at TIMESTAMP DEFAULT NOW()
);
